### NUL 扫描脚本

```bash
python3 - <<'EOF'
import os
hits=[]
for root,dirs,files in os.walk('.'):
    dirs[:]=[d for d in dirs if d not in ('node_modules','.git','dist','release','samples','invoices','vendor','tmp')]
    for fn in files:
        if not fn.endswith(('.ts','.js','.cjs','.mjs','.html','.css','.json','.md','.yml','.plist')): continue
        p=os.path.join(root,fn); d=open(p,'rb').read()
        if b'\x00' in d: hits.append((p,d.count(b'\x00')))
print('NUL files:',hits)
EOF
```
